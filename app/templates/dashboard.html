{% set endpoints = [ ("Chat Completions", "POST /v1/chat/completions", "ab", "OpenAI-compatible endpoint for generating chat responses."), ("Embeddings", "POST /v1/embeddings", "ab", "Generate vector embeddings for text inputs."), ("Rerank", "POST /v1/rerank", "av", "Re-order documents by relevance to a query."), ("Score", "POST /v1/score", "av", "Get relevance scores for query-document pairs.") ] %} {# Chat Completions #}
Chat Completions POST /v1/chat/completions

OpenAI-compatible endpoint for generating chat responses.

from openai import OpenAI

client = OpenAI(
    base_url="{{ gateway_base }}",
    api_key="{{ user.api_key }}"
)

resp = client.chat.completions.create(
    model="radllm",
    messages=[{"role": "user", "content": "Hello!"}]
)
{# Embeddings #}
Embeddings POST /v1/embeddings

Generate vector embeddings for text inputs.

from openai import OpenAI

client = OpenAI(
    base_url="{{ gateway_base }}",
    api_key="{{ user.api_key }}"
)

resp = client.embeddings.create(
    model="bge-m3",
    input=["The quick brown fox"]
)
{# Rerank #}
Rerank POST /v1/rerank

Re-order documents by relevance to a query.

import requests

resp = requests.post(
    "{{ gateway_base }}/rerank",
    headers={"Authorization": "Bearer {{ user.api_key }}"},
    json={
        "model": "bge-reranker-v2-m3",
        "query": "What is deep learning?",
        "documents": ["Doc A...", "Doc B..."],
        "top_n": 3
    }
)
